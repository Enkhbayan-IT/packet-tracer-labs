! Lab 03 - R1 Router-on-a-Stick and Extended ACL
! Reference configuration based on verified lab screenshots
hostname R1
!
interface GigabitEthernet0/0
 description TRUNK_TO_SW1
 no ip address
 no shutdown
!
interface GigabitEthernet0/0.10
 description HR_VLAN10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip access-group HR_POLICY in
!
interface GigabitEthernet0/0.20
 description IT_VLAN20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
!
ip access-list extended HR_POLICY
 deny icmp 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255
 permit ip any any
!
end
